# Build a real consumer from the published matrix. The table is the source of
# the symbol list so adding a promise to the documentation also tests it in CI.
set(matrix "${CMAKE_CURRENT_LIST_DIR}/../../docs/api-support.md")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${matrix}")
file(STRINGS "${matrix}" rows REGEX "^\\|" ENCODING UTF-8)
list(POP_FRONT rows header separator)
set(platforms STM32H743 STM32L496 PC ZYNQ7000)
list(FIND platforms "${UNI_HAL_TARGET_MCU}" column)
string(JOIN " | " headings ${platforms})
if(column EQUAL -1 OR NOT header STREQUAL "| Prefix | Function suffixes | ${headings} |")
    message(FATAL_ERROR "Unknown target or unexpected API matrix headings: ${header}")
endif()

set(references "")
foreach(row IN LISTS rows)
    if(NOT row MATCHES [[^\| `([a-z0-9_]+)` \| `([a-z0-9_, ]+)` \| ([+-]) \| ([+-]) \| ([+-]) \| ([+-]) \|$]])
        message(FATAL_ERROR "Malformed API matrix row: ${row}")
    endif()
    set(prefix "${CMAKE_MATCH_1}")
    string(REPLACE ", " ";" functions "${CMAKE_MATCH_2}")
    set(support "${CMAKE_MATCH_3}" "${CMAKE_MATCH_4}" "${CMAKE_MATCH_5}" "${CMAKE_MATCH_6}")
    list(GET support ${column} supported)
    if(supported STREQUAL "+")
        foreach(suffix IN LISTS functions)
            string(APPEND references "    LINK_API(uni_hal_${prefix}_${suffix});\n")
        endforeach()
    endif()
endforeach()
configure_file(client.c.in client.c @ONLY)

if(UNI_HAL_TARGET_MCU STREQUAL "PC")
    add_executable(public_api_link)
    target_link_libraries(public_api_link PRIVATE uni.hal)
    add_test(NAME public_api_link COMMAND public_api_link)
else()
    uni_hal_add_executable(public_api_link)
endif()
target_sources(public_api_link PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/client.c")
